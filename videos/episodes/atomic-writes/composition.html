Across trees

// Move a row between trees in one cross-tree saga.
await grainFactory
    .BeginAtomicWrite("rekey:order-42")
    .ForTree("orders-east")
        .Delete("order:42")
    .ForTree("orders-west")
        .Set("order:42", new Order("42", 250m))
    .CommitAsync(cancellationToken);

orders-eastDelete the old row

orders-westSet the new row

One decisionCommit both slices, or neither
